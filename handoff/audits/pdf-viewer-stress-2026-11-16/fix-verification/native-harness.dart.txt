class AuditKeys implements KeyManager {
  final bytes = Uint8List.fromList(List.generate(32, (i) => i + 1));
  @override
  bool get isUnlocked => true;
  @override
  Uint8List getMasterKey() => bytes;
  @override
  dynamic noSuchMethod(Invocation invocation) => super.noSuchMethod(invocation);
}

class AuditService extends DocumentService {
  AuditService({required super.database, required super.keyManager,
    required super.storage, required super.thumbnailService});
  int delayMs = 0;
  bool fail = false;
  int attempts = 0;
  final saves = <Map<String, Object?>>[];
  @override
  Future<void> saveAnnotatedDocument({required String documentId,
      required Uint8List annotatedPdfBytes}) async {
    attempts++;
    final event = <String, Object?>{'id': documentId, 'bytes': annotatedPdfBytes.length,
      'sha': DocumentCrypto.computeSha256(annotatedPdfBytes), 'start': DateTime.now().toIso8601String()};
    saves.add(event);
    if (delayMs > 0) await Future<void>.delayed(Duration(milliseconds: delayMs));
    if (fail) {event['failed'] = true; throw FileSystemException('Injected audit storage failure');}
    await super.saveAnnotatedDocument(documentId: documentId, annotatedPdfBytes: annotatedPdfBytes);
    event['complete'] = DateTime.now().toIso8601String();
  }
}

Future<Uint8List> pdf(int count, {bool mixed = false, bool scan = false}) async {
  final doc = pw.Document();
  for (var i = 0; i < count; i++) {
    final format = mixed && i.isOdd ? PdfPageFormat.a4.landscape : PdfPageFormat.a4;
    doc.addPage(pw.Page(pageFormat: format, build: (_) => pw.Column(
      crossAxisAlignment: pw.CrossAxisAlignment.start,
      children: scan ? [pw.Container(width: 300, height: 300, color: PdfColors.grey300)] : [
        pw.Text('PDF AUDIT PAGE ${i + 1}', style: const pw.TextStyle(fontSize: 24)),
        pw.SizedBox(height: 20),
        pw.Text('Alpha contract signature needle ${i + 1}', style: const pw.TextStyle(fontSize: 16)),
        for (var line = 0; line < 18; line++)
          pw.Text('Page ${i + 1} line $line: The quick brown fox reviews the document.', style: const pw.TextStyle(fontSize: 12)),
      ],
    )));
  }
  return doc.save();
}

Future<void> main() async {
  WidgetsFlutterBinding.ensureInitialized();
  SharedPreferences.setMockInitialValues({'setting_pdf_thumbnail_sidebar_open': true});
  root = await Directory.systemTemp.createTemp('qp-pdf-native-');
  final keys = AuditKeys();
  thumbnails = PdfThumbnailService(keyManager: keys, customDirectory: root);
  service = AuditService(database: AppDatabase.memory(), keyManager: keys,
    storage: DocumentLocalStorage(customDocumentsDirectory: Directory('${root.path}/documents'),
      customTempDirectory: Directory('${root.path}/temp')), thumbnailService: thumbnails);
  for (final entry in {'small': 4, 'large': 250, 'mixed': 12, 'scan': 3, 'stale': 4, 'corrupt': 1}.entries) {
    final bytes = entry.key == 'corrupt' ? Uint8List.fromList(utf8.encode('%PDF-1.7\ninvalid audit data'))
      : await pdf(entry.value, mixed: entry.key == 'mixed', scan: entry.key == 'scan');
    final created = await service.createDocumentFromPdfBytes(pdfBytes: bytes,
      pageCount: entry.key == 'stale' ? 1 : entry.value,
      title: 'PDF Audit ${entry.key}', source: DocumentSource.importedPdf, autoRunOcr: false);
    fixtures[entry.key] = await service.resolveDocument(created.document.id);
  }
  FlutterError.onError = (details) {
    errors.add({'error': details.exceptionAsString(), 'stack': details.stack.toString()});
    FlutterError.presentError(details);
  };
  ui.PlatformDispatcher.instance.onError = (error, stack) {
    errors.add({'error': error.toString(), 'stack': stack.toString()});
    stderr.writeln('$error\n$stack'); return true;
  };
  WidgetsBinding.instance.addTimingsCallback((frames) {
    for (final f in frames) {
      timings.add({'buildUs': f.buildDuration.inMicroseconds, 'rasterUs': f.rasterDuration.inMicroseconds,
        'totalUs': f.totalSpan.inMicroseconds});
    }
  });
  runApp(ProviderScope(overrides: [documentServiceProvider.overrideWithValue(service),
    pdfThumbnailServiceProvider.overrideWithValue(thumbnails)], child: RepaintBoundary(key: capture,
    child: MaterialApp(navigatorKey: nav, title: 'Quiet Paper PDF Audit — isolated test window',
      theme: AppTheme.light(), home: const Scaffold(body: Center(child: Text('PDF viewer audit — synthetic documents only')))))));
  final server = await HttpServer.bind(InternetAddress.loopbackIPv4, 38927);
  stdout.writeln('PDF_AUDIT_READY ${root.path}');
  await for (final request in server) {
    try {
      final input = request.method == 'POST' ? jsonDecode(await utf8.decoder.bind(request).join()) as Map<String, dynamic> : <String, dynamic>{};
      if (request.uri.path == '/open') {
        nav.currentState!.popUntil((route) => route.isFirst);
        await Future<void>.delayed(const Duration(milliseconds: 400));
        current = input['fixture'] as String? ?? 'small';
        final res = fixtures[current]!;
        nav.currentState!.push(MaterialPageRoute<void>(builder: (_) => DocumentViewerScreen(
          documentId: res.data!.documentId, title: res.data!.title,
          initialResolution: res, initialPageIndex: input['initialPage'] as int? ?? 0)));
        await Future<void>.delayed(const Duration(milliseconds: 1300));
      } else if (request.uri.path == '/config') {
        service.delayMs = input['delayMs'] as int? ?? 0;
        service.fail = input['fail'] as bool? ?? false;
      } else if (request.uri.path == '/add') {
        final controller = viewer()?.controller;
        final count = input['count'] as int? ?? 1;
        for (var i = 0; i < count; i++) {
          controller?.addAnnotation(StickyNoteAnnotation(icon: PdfStickyNoteIcon.note, pageNumber: input['page'] as int? ?? 1,
            text: 'Audit note ${controller.getAnnotations().length + 1}', position: Offset(70.0 + i % 20 * 5, 110.0 + i % 15 * 5)));
        }
      } else if (request.uri.path == '/jump') {
        viewer()?.controller?.jumpToPage(input['page'] as int);
        await Future<void>.delayed(const Duration(milliseconds: 450));
      } else if (request.uri.path == '/maybePop') {
        unawaited(nav.currentState!.maybePop());
        await Future<void>.delayed(const Duration(milliseconds: 600));
      } else if (request.uri.path == '/thumbnails') {
        final res = fixtures[current]!.data!;
        final count = input['count'] as int? ?? 40;
        final watch = Stopwatch()..start();
        final results = await Future.wait(List.generate(count, (i) => thumbnails.getPageThumbnail(
          documentId: res.documentId, pageIndex: i % res.pageCount, pdfBytes: res.pdfBytes)));
        request.response.headers.contentType = ContentType.json;
        request.response.write(jsonEncode({'count': count, 'nonNull': results.whereType<Uint8List>().length, 'ms': watch.elapsedMilliseconds}));
        await request.response.close(); continue;
      }
      if (request.uri.path == '/screenshot') {
        final boundary = capture.currentContext!.findRenderObject()! as RenderRepaintBoundary;
        final snapshot = await boundary.toImage(pixelRatio: 1);
        final bytes = await snapshot.toByteData(format: ui.ImageByteFormat.png);
        request.response.headers.contentType = ContentType('image', 'png');
        request.response.add(bytes!.buffer.asUint8List()); snapshot.dispose();
      } else {
        request.response.headers.contentType = ContentType.json;
        request.response.write(jsonEncode(inspect()));
      }
    } catch (error, stack) {
      request.response.statusCode = 500;
      request.response.write(jsonEncode({'error': error.toString(), 'stack': stack.toString()}));
    }
    await request.response.close();
  }
}

SfPdfViewer? viewer() {
  SfPdfViewer? found;
  void visit(Element element) {
    if (element.widget is SfPdfViewer) found = element.widget as SfPdfViewer;
    element.visitChildren(visit);
  }
  WidgetsBinding.instance.rootElement?.visitChildren(visit);
  return found;
}

List<double>? rect(Element element) {
  try {
    final render = element.findRenderObject();
    if (render is! RenderBox || !render.attached || !render.hasSize) return null;
    final point = render.localToGlobal(Offset.zero);
    final values = [point.dx, point.dy, render.size.width, render.size.height];
    return values.every((value) => value.isFinite) ? values : null;
  } catch (_) { return null; }
}

Map<String, Object?> inspect() {
  final labels = <Map<String, Object?>>[];
  final fields = <Map<String, Object?>>[];
  final tiles = <Map<String, Object?>>[];
  void visit(Element element) {
    final widget = element.widget;
    if (widget is Text) labels.add({'kind': 'text', 'text': widget.data ?? widget.textSpan?.toPlainText(), 'rect': rect(element)});
    if (widget is Tooltip) labels.add({'kind': 'tooltip', 'text': widget.message, 'rect': rect(element)});
    if (widget is Semantics && widget.properties.label != null) tiles.add({'text': widget.properties.label, 'selected': widget.properties.selected, 'rect': rect(element)});
    if (widget is EditableText) fields.add({'text': widget.controller.text, 'focus': widget.focusNode.hasFocus, 'rect': rect(element)});
    element.visitChildren(visit);
  }
  WidgetsBinding.instance.rootElement?.visitChildren(visit);
  final v = viewer();
  final c = v?.controller;
  return {'fixture': current, 'viewer': v == null ? null : {'page': c?.pageNumber, 'pages': c?.pageCount,
    'zoom': c?.zoomLevel, 'annotations': c?.getAnnotations().map((a) => {'type': a.runtimeType.toString(), 'page': a.pageNumber}).toList(),
    'layout': v.pageLayoutMode.toString(), 'interaction': v.interactionMode.toString(),
    'mode': c?.annotationMode.toString(), 'initial': v.initialPageNumber},
    'labels': labels, 'fields': fields, 'tiles': tiles, 'errors': errors, 'saves': service.saves,
    'attempts': service.attempts, 'root': root.path, 'timings': timings.length,
    'slowFrames': timings.where((f) => (f['totalUs'] as int) > 100000).toList()};
}
