class GateRenderer implements PdfPageRenderer {
  final entered = Completer<void>();
  final release = Completer<void>();
  int calls = 0;
  @override
  Future<RenderedPage?> renderSinglePage(Uint8List pdfBytes, int pageIndex, {double dpi = 150}) async {
    calls++;
    if (!entered.isCompleted) entered.complete();
    await release.future;
    return RenderedPage(pageNumber: pageIndex + 1, imageBytes: Uint8List.fromList([137, 80, 78, 71, pdfBytes.first]), width: 1, height: 1);
  }
  @override
  dynamic noSuchMethod(Invocation invocation) => super.noSuchMethod(invocation);
}
