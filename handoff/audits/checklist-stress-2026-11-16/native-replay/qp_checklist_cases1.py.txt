from qp_checklist_driver import *
CHECKS=pathlib.Path('/tmp/qp-checklist-native-checks.jsonl')
def evaluate(step,expected,predicate):
    s=record(step)
    item={'case':s['case'],'step':step,'expected':expected,'pass':bool(predicate(s)),'errorCount':len(s['errors'])}
    with CHECKS.open('a') as f:f.write(json.dumps(item)+'\n')
    print('CHECK',json.dumps(item),flush=True)
    if s['errors']:raise RuntimeError('Stop after runtime failure; restart isolated process')
    return s

def blocks(s):return s['rich'][0]['document']['blocks']
def texts(s):return [''.join(span['text'] for span in b.get('spans',[])) for b in blocks(s)]
def choose(name):
    label('Insert Options (Table, Link, Media)','tooltip');time.sleep(.8);label(name,'text');time.sleep(.8)

for offset in [0,4,9]:
    reset('visual','- [ ] AlphaBeta\n- [ ] Neighbor');field(text='AlphaBeta');key('Home')
    for _ in range(offset):key('Right')
    key('Return')
    evaluate('Visual Enter at offset '+str(offset),'split exact content and focus new unchecked task',lambda s:texts(s)==['AlphaBeta'[:offset],'AlphaBeta'[offset:],'Neighbor'] and any(f['focus'] and f['text']=='AlphaBeta'[offset:] for f in s['fields']))

for reversed in [False,True]:
    reset('visual','- [ ] AlphaBeta\n- [ ] Neighbor');field(text='AlphaBeta');key('Home')
    for _ in range(7 if reversed else 2):key('Right')
    for _ in range(5):key('Left' if reversed else 'Right','SHIFT')
    record('Visual selected range before Enter reversed='+str(reversed));key('Return')
    evaluate('Visual Enter replaces selected range reversed='+str(reversed),'selected phaBe is removed, producing Al and ta',lambda s:texts(s)==['Al','ta','Neighbor'])

for initial,target in [('- [ ] First\n- [ ] Second','Second'),('Paragraph\n- [ ] Second','Second'),('- [ ] First\n- [ ] ',''),('- [ ] ','')]:
    reset('visual',initial);field(text=target);key('Home');key('BackSpace')
    evaluate('Visual Backspace start '+repr(initial),'all letters survive and a live field retains focus',lambda s:''.join(texts(s))==initial.replace('- [ ] ','').replace('\n','') and any(f['focus'] for f in s['fields']))

reset('visual','- [ ] First\n- [ ] Second');field(text='First');key('End');key('Delete')
evaluate('Visual forward Delete at row end','joins First and Second in one task',lambda s:texts(s)==['FirstSecond'])

reset('visual','- [ ] First\n- [ ] ');field(text='');type_text('   ');key('Return')
evaluate('Visual whitespace-only task Enter','exits to paragraph and keeps First',lambda s:blocks(s)[-1]['type']=='paragraph' and texts(s)[0]=='First')

reset('visual','- [ ] Parent\n  - [ ] Child\n    - [x] Grandchild\n- [ ] Sibling');field(text='Sibling');key('End');type_text('x')
s=evaluate('Visual edit nested checklist then save','child indentation remains in canonical representation',lambda s:'indent' in json.dumps(blocks(s)[1]) and blocks(s)[1].get('indent')==1)
record('Visual nested checklist canonical Markdown')

reset('visual','- [ ] Task ');field(text='Task ');key('End');choose('Tag');type_text('nativeaudit')
evaluate('Visual Tag insertion inside checklist row','saves Task #nativeaudit',lambda s:texts(s)==['Task #nativeaudit'])
print('BATCH1 COMPLETE',flush=True)
