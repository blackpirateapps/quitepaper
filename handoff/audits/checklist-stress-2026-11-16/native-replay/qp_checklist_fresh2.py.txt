prepare()
reset('visual','- [ ] Seed');field(text='Seed');key('End')
for i in range(1,41):
    key('Return');type_text('t'+str(i))
    if i%10==0:record('fresh process rapid creation '+str(i+1)+' tasks')
evaluate('Fresh process 41 task rapid native creation','all 41 expected task labels retained',lambda s:texts(s)==['Seed']+['t'+str(i) for i in range(1,41)])

reset('visual','- [ ] Seed');field(text='Seed');key('End')
for i in range(1,31):
    key('Return');time.sleep(.4);type_text('t'+str(i));time.sleep(.15)
    if i%10==0:record('paced creation '+str(i+1)+' tasks')
evaluate('Paced creation control 31 tasks','all 31 expected task labels retained',lambda s:texts(s)==['Seed']+['t'+str(i) for i in range(1,31)])

reset('visual','- [ ] Alpha\n- [ ] Neighbor');field(text='Alpha');key('End');api('/clipboard',{'text':'\n- [ ] Beta\n- [x] Gamma'});key('v','CTRL');time.sleep(4)

saved=s['savedContent'];reset('visual',saved)
evaluate('Reopen native checklist paste saved content','all pasted task labels survive reopening',lambda s:texts(s)==['Alpha','Beta','Gamma','Neighbor'])

reset('super','- [ ] Parent\n  - [ ] Child\n    - [x] Grandchild\n- [ ] Sibling')
s=record('Fresh Super nested parse before any field edit');r=next(t['rect'] for t in s['tasks'] if t['text']=='Sibling')
click(r,.35);key('End');type_text('x');time.sleep(1)
s=evaluate('Fresh Super nested task save','four tasks retained, including checked grandchild',lambda s:len(s['super'][0]['nodes'])==4 and 'Grandchild' in s['savedContent'])
print('SUPER_NESTED_SAVED',repr(s['savedContent']),flush=True)
print('FRESH2 COMPLETE',flush=True)
