import subprocess,json,time,pathlib,urllib.request,sys
BASE='http://127.0.0.1:38918'
def api(path='/state'):
 return json.load(urllib.request.urlopen(BASE+path,timeout=10))
def dispatch(expr):
 r=subprocess.run(['hyprctl','eval','hl.dispatch('+expr+')'],capture_output=True,text=True,check=True)
 if 'error:' in r.stdout:raise RuntimeError(r.stdout)
clients=json.loads(subprocess.check_output(['hyprctl','-j','clients']))
root_pid=json.loads(pathlib.Path('/tmp/qp-swipe-audit/process.json').read_text())['pid']
def owned(pid):
 while pid>1:
  if pid==root_pid:return True
  try:pid=int(pathlib.Path('/proc/'+str(pid)+'/stat').read_text().split(') ')[1].split()[1])
  except (OSError,ValueError):return False
 return False
for attempt in range(50):
 clients=json.loads(subprocess.check_output(['hyprctl','-j','clients']))
 candidates=[c for c in clients if owned(c['pid']) and c['mapped']]
 if candidates:break
 time.sleep(.1)
else:raise RuntimeError('Audit process has no mapped window')
c=candidates[0]
W='address:'+c['address']
width=int(sys.argv[2]) if len(sys.argv)>2 else 420
for expr in ['hl.dsp.window.float({window='+json.dumps(W)+',action="enable"})','hl.dsp.window.resize({window='+json.dumps(W)+',x='+str(width)+',y=800,relative=false})','hl.dsp.window.move({window='+json.dumps(W)+',x=80,y=45,relative=false})','hl.dsp.focus({window='+json.dumps(W)+'})']:
 dispatch(expr)
time.sleep(.6)
c=next(c for c in json.loads(subprocess.check_output(['hyprctl','-j','clients'])) if 'address:'+c['address']==W)
pointer=subprocess.Popen(['/tmp/qp-swipe-audit/pointer'],stdin=subprocess.PIPE,stdout=subprocess.PIPE,text=True)
assert pointer.stdout.readline().strip()=='READY'
def send(cmd):
 pointer.stdin.write(cmd+'\n');pointer.stdin.flush();assert pointer.stdout.readline().strip()=='OK'
def motion(dx,dy):send('m '+str(dx)+' '+str(dy))
def move(x,y):dispatch('hl.dsp.cursor.move({x='+str(round(c['at'][0]+x))+',y='+str(round(c['at'][1]+y))+'})')
def button(state):send('b '+('1' if state=='down' else '0'))
def click(x,y):move(x,y);time.sleep(.03);button('down');time.sleep(.03);button('up');time.sleep(.04)
def back():
 s=api();choices=[l for l in s['labels'] if l.get('tooltip')=='Back' and l.get('rect')];r=choices[-1]['rect'];click(r[0]+r[2]/2,r[1]+r[3]/2)
def screenshot(name):pathlib.Path('/tmp/qp-swipe-audit/'+name+'.png').write_bytes(urllib.request.urlopen(BASE+'/screenshot').read())
def drag(name,deltas,earlyback=False):
 s=api();r=s['pulls'][0]['rect'];x=r[0]+r[2]*.65
 api('/record');move(x,300);button('down');time.sleep(.05);y=300
 for dy in deltas:
  y+=dy;motion(0,dy);time.sleep(.035)
 hold=api();screenshot(name+'-held');button('up')
 if earlyback:
  time.sleep(.04);back()
 time.sleep(1.5);end=api();frames=api('/frames')
 pathlib.Path('/tmp/qp-swipe-audit/'+name+'.json').write_text(json.dumps({'held':hold,'end':end,'frames':frames}))
 def brief(s):
  notes=[l for l in s['labels'] if l.get('text')=='Swipe audit note 0'];return {'pull':s['pulls'],'scroll':[x['pixels'] for x in s['scrolls']],'note':notes,'search':s['searchCount'],'errors':s['errors']}
 print(name,json.dumps({'held':brief(hold),'end':brief(end)}),flush=True)
 screenshot(name+'-end')
 return end
if __name__=='__main__':
 prefix=sys.argv[1]
 drag(prefix+'-short',[20]*4)
 drag(prefix+'-full',[25]*10)
 back();time.sleep(.7)
 drag(prefix+'-reverse',[25]*10+[-25]*8)
 if any(s['pixels']>0 for s in api()['scrolls']):
  r=api()['pulls'][0]['rect'];move(r[0]+r[2]*.65,250);button('down');time.sleep(.04)
  for _ in range(12):motion(0,25);time.sleep(.03)
  button('up');time.sleep(.8)
 drag(prefix+'-early-back',[25]*10,True)
