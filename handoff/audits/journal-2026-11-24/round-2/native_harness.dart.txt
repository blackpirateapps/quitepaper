final captureKey = GlobalKey();
final errors = <String>[];
late ProviderContainer container;
late DriftNotesRepository repository;
Completer<http.Response>? weatherResponse;
var weatherRequests = 0;

Future<void> main() async {
  WidgetsFlutterBinding.ensureInitialized();
  QuietPaperLinuxSharePlugin.register();
  await initializeDateFormatting('bn');
  // This entrypoint is an isolated audit fixture, never a production target.
  // ignore: invalid_use_of_visible_for_testing_member
  SharedPreferences.setMockInitialValues({
    'onboarding_completed_v1': true,
    'onboarding_tour_seen_v1': true,
    'setting_show_place_and_weather_on_entries': false,
    'setting_suggest_place_for_past_entries': false,
  });
  final prefs = await SharedPreferences.getInstance();
  final database = AppDatabase.memory();
  repository = DriftNotesRepository(database);
  container = ProviderContainer(overrides: [
    databaseProvider.overrideWithValue(database),
    notesRepositoryProvider.overrideWithValue(repository),
    sharedPreferencesProvider.overrideWithValue(prefs),
    currentDestinationProvider.overrideWith((ref) => AppDestination.allJournalEntries),
  ]);
  FlutterError.onError = (details) {
    errors.add(details.toString());
    FlutterError.presentError(details);
  };
  ui.PlatformDispatcher.instance.onError = (error, stack) {
    errors.add('$error\n$stack');
    return true;
  };
  registerExtension('ext.journalAudit', (method, args) async {
    try {
      final result = await action(args);
      return ServiceExtensionResponse.result(jsonEncode(result));
    } catch (e, s) {
      return ServiceExtensionResponse.result(jsonEncode({'error': '$e', 'stack': '$s'}));
    }
  });
  runApp(UncontrolledProviderScope(container: container,
    child: RepaintBoundary(key: captureKey, child: const QuietPaperApp())));
}

List<double>? rect(Element e) {
  try {
    final r = e.findRenderObject();
    if (r is! RenderBox || !r.attached || !r.hasSize) return null;
    final o = r.localToGlobal(Offset.zero);
    return [o.dx, o.dy, r.size.width, r.size.height];
  } catch (_) { return null; }
}

Future<Map<String, Object?>> action(Map<String, String> args) async {
  final command = args['action'] ?? 'snapshot';
  if (command == 'deferWeather') {
    weatherResponse = Completer<http.Response>();
    return http.runWithClient(() => action({'action': 'tap', 'text': 'Fetch Weather'}),
      () => MockClient((request) { weatherRequests++; return weatherResponse!.future; }));
  }
  if (command == 'seed') {
    final count = int.parse(args['count'] ?? '35');
    final now = DateTime.now();
    await container.read(databaseProvider).transaction(() async {
    for (var i = 0; i < count; i++) {
      final d = DateTime(now.year, now.month, now.day - i);
      final note = await repository.getOrCreateJournalEntry(d);
      final key = JournalDateHelper.toDateString(d);
      final address = i.isEven ? 'Kyoto, Kyoto, Japan' : 'Kolkata, West Bengal, India';
      await repository.saveNote(note.copyWith(title: 'Audit day $key',
        content: '---\njournal: true\ndate: $key\nmoment: ${i % 3 == 0 ? 'travel' : 'morning'}\nlocation:\n  address: "$address"\n  latitude: 1.0\n  longitude: 2.0\n---\n\n${List.filled(i % 4 + 1, 'Synthetic journal reflection. ').join()}'));
      if (i % 100 == 0) stdout.writeln('AUDIT_SEED_PROGRESS=$i');
    }
    for (var y = 1; y <= 7; y++) {
      await repository.getOrCreateJournalEntry(DateTime(now.year - y, now.month, now.day));
    }
    });
  } else if (command == 'completeWeather') {
    weatherResponse!.complete(http.Response('{"current":{"temperature_2m":25,"weather_code":3}}', 200));
  } else if (command == 'coordinate') {
    final note = await repository.getOrCreateJournalEntry(DateTime(2026, 1, 15));
    await repository.saveNote(note.copyWith(content: '---\njournal: true\ndate: 2026-01-15\nlocation:\n  address: ""\n  latitude: 22.57\n  longitude: 88.36\n---\n\nCoordinate-only entry.'));
  } else if (command == 'longTitles') {
    final notes = await repository.getAllJournalEntries();
    await container.read(databaseProvider).transaction(() async {
      for (final note in notes) {
        await repository.saveNote(note.copyWith(title: '${note.journalDate} ${List.filled(6, 'A long reflection on this memorable journal day').join(' ')}'));
      }
    });
  } else if (command == 'resume') {
    WidgetsBinding.instance.handleAppLifecycleStateChanged(AppLifecycleState.inactive);
    WidgetsBinding.instance.handleAppLifecycleStateChanged(AppLifecycleState.resumed);
  } else if (command == 'mergeFixture') {
    await container.read(databaseProvider).transaction(() async {
      for (var i = 0; i < 3; i++) {
        final date = DateTime(2026, 11, 20 + i);
        final key = JournalDateHelper.toDateString(date);
        final city = ['Alpha', 'Beta', 'Gamma'][i];
        final note = await repository.getOrCreateJournalEntry(date);
        await repository.saveNote(note.copyWith(title: '$city journal', content: '---\njournal: true\ndate: $key\nlocation:\n  address: "$city, India"\n  latitude: 1.0\n  longitude: 2.0\n---\n\nSynthetic entry for $city.'));
      }
    });
  } else if (command == 'historyFixture') {
    await repository.getOrCreateJournalEntry(DateTime(2025, 11, 24));
  } else if (command == 'olderBackfill') {
    await repository.getOrCreateJournalEntry(DateTime(2010, 11, 23));
  } else if (command == 'destination') {
    rootNavigatorKey.currentState?.popUntil((r) => r.isFirst);
    container.read(currentDestinationProvider.notifier).state = AppDestination.values.byName(args['value']!);
  } else if (command == 'month') {
    container.read(calendarVisibleMonthProvider.notifier).state = (year: int.parse(args['year']!), month: int.parse(args['month']!));
  } else if (command == 'select') {
    container.read(calendarSelectedDateProvider.notifier).state = args['date'];
  } else if (command == 'bengali') {
    await container.read(defaultSettingsProvider.notifier).setBengaliDates(args['value'] == 'true');
  } else if (command == 'trash' || command == 'restore') {
    final note = await repository.getJournalEntry(args['date']!);
    if (command == 'trash') { await repository.trashNote(note!.id); }
    else { await repository.restoreFromTrash(note!.id); }
  } else if (command == 'tap' || command == 'longPress') {
    Element? target;
    void visit(Element e) {
      final w = e.widget;
      if ((args['text'] != null && w is Text && w.data == args['text']) ||
          (args['tooltip'] != null && w is Tooltip && w.message == args['tooltip']) ||
          (args['semantics'] != null && w is Semantics && w.properties.label == args['semantics'])) {
        target ??= e;
      }
      e.visitChildren(visit);
    }
    WidgetsBinding.instance.rootElement!.visitChildren(visit);
    final box = target == null ? null : rect(target!);
    if (box == null && args['x'] == null) throw StateError('Tap target not found: $args');
    final position = args['x'] != null ? Offset(double.parse(args['x']!), double.parse(args['y']!))
      : Offset(box![0] + box[2] / 2, box[1] + box[3] / 2);
    GestureBinding.instance.handlePointerEvent(PointerDownEvent(pointer: 41, position: position));
    if (command == 'longPress') await Future<void>.delayed(const Duration(milliseconds: 700));
    GestureBinding.instance.handlePointerEvent(PointerUpEvent(pointer: 41, position: position));
  } else if (command == 'back') {
    rootNavigatorKey.currentState?.pop();
  } else if (command == 'screenshot') {
    await WidgetsBinding.instance.endOfFrame;
    final boundary = captureKey.currentContext!.findRenderObject()! as RenderRepaintBoundary;
    final img = await boundary.toImage(pixelRatio: 1);
    final data = await img.toByteData(format: ui.ImageByteFormat.png);
    await File(args['path']!).writeAsBytes(data!.buffer.asUint8List());
    img.dispose();
  }
  await Future<void>.delayed(const Duration(milliseconds: 350));
  final labels = <Map<String, Object?>>[];
  final tiles = <Map<String, Object?>>[];
  final fields = <Map<String, Object?>>[];
  void visit(Element e) {
    final w = e.widget;
    if (w is Text && w.data != null) labels.add({'text': w.data, 'rect': rect(e)});
    if (w is Tooltip) labels.add({'tooltip': w.message, 'rect': rect(e)});
    if (w is EditableText) fields.add({'text': w.controller.text, 'rect': rect(e)});
    if (w is JournalTimelineTile) tiles.add({'date': w.note.journalDate, 'rect': rect(e)});
    e.visitChildren(visit);
  }
  WidgetsBinding.instance.rootElement!.visitChildren(visit);
  final notes = await repository.getAllJournalEntries();
  return {'action': command, 'labels': labels, 'tiles': tiles, 'fields': fields,
    'clock': DateTime.now().toIso8601String(),
    'weatherRequests': weatherRequests,
    if (command == 'historyStatus') 'earliestDbYear': await repository.getEarliestJournalYear(),
    if (command == 'historyStatus') 'earliestCachedYear': container.read(earliestJournalYearFutureProvider).valueOrNull,
    if (command == 'historyStatus') 'hasMoreOlderYears': container.read(historicalWeekProvider).hasMoreOlderYears,
    'actualToday': JournalDateHelper.todayString(),
    'cachedToday': container.read(todayJournalDateProvider),
    'cachedHistoryReference': container.read(historicalReferenceDateProvider).toIso8601String(),
    'aliases': container.read(placeAliasStoreProvider).map((k, v) => MapEntry(k, {'canonicalKey': v.canonicalKey, 'displayName': v.displayName})),
    'destination': container.read(currentDestinationProvider).name,
    'month': container.read(calendarVisibleMonthProvider).toString(),
    'selected': container.read(calendarSelectedDateProvider),
    'count': notes.length, 'notes': notes.map((n) => {'date': n.journalDate, 'title': n.title, 'tags': n.tags}).toList(),
    'errors': errors};
}
